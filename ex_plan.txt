This script detects embedded images (the original JPGs) and OCRs them directly, which is faster than rendering the whole page.pythonimport fitz  # PyMuPDF
import pytesseract
from PIL import Image
import io

# POINT 1: Connect Python to the Tesseract Engine
pytesseract.pytesseract.tesseract_cmd = r'C:\Program Files\Tesseract-OCR\tesseract.exe'

def extract_text_from_image_pdf(pdf_path):
    doc = fitz.open(pdf_path)
    full_text = ""

    # POINT 2: Loop through each page (Efficient for Large PDFs)
    for page_index, page in enumerate(doc):
        
        # Check for embedded images (The "Merged JPG" scenario)
        image_list = page.get_images(full=True)
        
        if image_list:
            print(f"Found {len(image_list)} images on page {page_index + 1}")
            
            for img_index, img in enumerate(image_list):
                xref = img[0]
                base_image = doc.extract_image(xref)
                image_bytes = base_image["image"]
                
                # Load image from bytes
                image = Image.open(io.BytesIO(image_bytes))
                
                # POINT 3: Perform OCR
                # text = pytesseract.image_to_string(image, lang='eng') # English
                # For better accuracy with blocks of text, use page segmentation mode 6:
                text = pytesseract.image_to_string(image, config='--psm 6')
                
                full_text += f"\n--- Page {page_index + 1} Image {img_index + 1} ---\n" + text
        else:
            # Fallback: If no embedded image found, render the whole page as an image
            pix = page.get_pixmap()
            img = Image.frombytes("RGB", [pix.width, pix.height], pix.samples)
            text = pytesseract.image_to_string(img)
            full_text += f"\n--- Page {page_index + 1} ---\n" + text

    return full_text

MODULES TO INCLUDE:

DEFINE TESSERACT_PATH = "C:\Program Files\Tesseract-OCR\tesseract.exe"
DEFINE BENGALI_LANG_CODE = "ben"

CONFIGURE pytesseract.tesseract_cmd = TESSERACT_PATH

2. CORE PROCESSING LOGIC (The Engine)

FUNCTION extract_bengali_pdf_to_word(input_pdf_path, output_word_path):
    
    // Initialize the Word Document object (Acts as our local storage/book)
    INITIALIZE word_doc = docx.Document()
    ADD_HEADING(word_doc, text="Extracted Book Content", level=0)
    
    // Open the large PDF file
    OPEN pdf_file = fitz.open(input_pdf_path)
    TOTAL_PAGES = LENGTH(pdf_file)
    
    // Loop through the PDF page by page to conserve RAM
    FOR each PAGE_INDEX from 0 to TOTAL_PAGES - 1:
        PRINT "Processing Page " + (PAGE_INDEX + 1)
        
        // Add a visual page marker in the Word Document
        ADD_PARAGRAPH(word_doc, text="--- PAGE " + (PAGE_INDEX + 1) + " ---")
        
        // Step A: Extract all embedded JPG images from this specific page
        IMAGE_LIST = pdf_file.get_page_images(PAGE_INDEX)
        
        IF IMAGE_LIST is NOT EMPTY:
            FOR each IMAGE_INFO in IMAGE_LIST:
                XREF = IMAGE_INFO[0]
                RAW_IMAGE_DATA = pdf_file.extract_image(XREF)
                IMAGE_BYTES = RAW_IMAGE_DATA["image"]
                
                // Convert bytes directly to an Image object in memory
                PIL_IMAGE = Image.open(io.BytesIO(IMAGE_BYTES))
                
                // Step B: Send image to Tesseract optimized for Bengali script
                // We use Page Segmentation Mode (--psm 6 or 3) for book-like layouts
                EXTRACTED_TEXT = pytesseract.image_to_string(
                    PIL_IMAGE, 
                    lang=BENGALI_LANG_CODE, 
                    config="--psm 3"
                )
                
                // Step C: Clean text and save line-by-line into Word
                SPLIT_LINES = EXTRACTED_TEXT.split('\n')
                FOR each LINE in SPLIT_LINES:
                    IF LINE is not empty or white space:
                        ADD_PARAGRAPH(word_doc, text=LINE.strip())
                        
        ELSE:
            // FALLBACK STRATEGY: If page has no embedded JPGs, it might be a flat scan.
            // Render the entire page canvas to a high-res image (300 DPI)
            PIXMAP = pdf_file[PAGE_INDEX].get_pixmap(dpi=300)
            PIL_IMAGE = Image.frombytes("RGB", [PIXMAP.width, PIXMAP.height], PIXMAP.samples)
            
            EXTRACTED_TEXT = pytesseract.image_to_string(PIL_IMAGE, lang=BENGALI_LANG_CODE)
            
            SPLIT_LINES = EXTRACTED_TEXT.split('\n')
            FOR each LINE in SPLIT_LINES:
                IF LINE is not empty:
                    ADD_PARAGRAPH(word_doc, text=LINE.strip())
                    
        // CRITICAL STEP FOR LARGE FILES: Save progress incrementally to prevent data loss
        SAVE word_doc to output_word_path
        
    CLOSE pdf_file
    PRINT "Extraction complete. Document saved successfully."

--------------------------------------------------------------------------------

FUNCTION build_user_interface():
    INITIALIZE Window = custom_tkinter.CTkWindow()
    SET Window.title = "Bengali PDF Book OCR Extractor"
    SET Window.geometry = "600x400"
    
    // UI Variables
    VARIABLE selected_pdf_path = ""
    VARIABLE save_destination_path = ""
    
    // UI Elements
    CREATE BUTTON "Select Source PDF" -> ON_CLICK: open_file_dialog() -> updates selected_pdf_path
    CREATE BUTTON "Select Word Save Location" -> ON_CLICK: save_file_dialog() -> updates save_destination_path
    
    CREATE PROGRESS_BAR to show conversion status (Page X of Y)
    
    CREATE BUTTON "START EXTRACTION" -> ON_CLICK:
        IF selected_pdf_path is empty OR save_destination_path is empty:
            SHOW_ERROR("Please select files first!")
        ELSE:
            DISABLE UI BUTTONS (Prevents freezing/double-clicking)
            RUN extract_bengali_pdf_to_word(selected_pdf_path, save_destination_path) in a BACKGROUND THREAD
            ENABLE UI BUTTONS
            SHOW_SUCCESS("Your Word book is ready!")

    RUN Window.mainloop()

START build_user_interface()
